        function updateBannerSlides(index) {
            currentSlide = (index + totalSlides) % totalSlides;

            desktopSlides.forEach((slide, idx) => {
                if (idx === currentSlide) {
                    slide.classList.remove('opacity-0', 'invisible');
                    slide.classList.add('opacity-100', 'visible');
                } else {
                    slide.classList.remove('opacity-100', 'visible');
                    slide.classList.add('opacity-0', 'invisible');
                }
            });

            mobileSlides.forEach((slide, idx) => {
                if (idx === currentSlide) {
                    slide.classList.remove('opacity-0', 'invisible');
                    slide.classList.add('opacity-100', 'visible');
                } else {
                    slide.classList.remove('opacity-100', 'visible');
                    slide.classList.add('opacity-0', 'invisible');
                }
            });

            // Cập nhật chấm chỉ báo
            for (let i = 0; i < totalSlides; i++) {
                const dot = document.getElementById('dot-' + i);
                if (dot) {
                    if (i === currentSlide) {
                        dot.className = 'w-6 h-2 rounded-full bg-brand-blue transition-all';
                    } else {
                        dot.className = 'w-2.5 h-2 rounded-full bg-slate-300 transition-all';
                    }
                }
            }
        }

        function nextSlide() {
            updateBannerSlides(currentSlide + 1);
